-- ============================================================
-- Investment Journal Database Schema
-- Supabase / PostgreSQL
--
-- 기존 테이블과 데이터를 삭제하지 않습니다.
-- 예제 데이터도 삽입하지 않습니다.
-- ============================================================


-- ============================================================
-- 1. Profiles
-- Supabase Auth 사용자와 연결되는 사용자 프로필
-- ============================================================

create table if not exists public.profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  display_name text,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);


-- ============================================================
-- 2. Journal Entries
-- 하루 단위 투자 일지
-- ============================================================

create table if not exists public.journal_entries (
  id bigint generated by default as identity primary key,

  user_id uuid not null references auth.users(id) on delete cascade,

  journal_date date not null,

  market_summary text,

  new_learnings text,
  changed_thoughts text,
  most_important_change text,

  judgment_action text,
  judgment_reason text,
  confidence smallint,

  tomorrow_checks text,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),

  constraint journal_entries_action_check
    check (
      judgment_action is null
      or judgment_action in (
        '매수',
        '추가매수',
        '보유',
        '관망',
        '일부매도',
        '매도'
      )
    ),

  constraint journal_entries_confidence_check
    check (
      confidence is null
      or confidence between 1 and 5
    ),

  constraint journal_entries_user_date_unique
    unique (user_id, journal_date)
);


-- ============================================================
-- 3. Market Indicators
-- 하루의 주요 시장 지표
-- ============================================================

create table if not exists public.market_indicators (
  id bigint generated by default as identity primary key,

  journal_id bigint not null
    references public.journal_entries(id) on delete cascade,

  indicator_name text not null,
  value numeric,
  change_value numeric,
  change_percent numeric,
  meaning text,

  created_at timestamptz not null default now()
);


-- ============================================================
-- 4. Stocks
-- 사용자가 관심/보유하는 종목
-- ============================================================

create table if not exists public.stocks (
  id bigint generated by default as identity primary key,

  user_id uuid not null
    references auth.users(id) on delete cascade,

  ticker text,
  market text,
  company_name text not null,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),

  constraint stocks_user_company_unique
    unique (user_id, company_name)
);


-- ============================================================
-- 5. Investment Theses
-- 종목별 장기 투자 가설
-- ============================================================

create table if not exists public.investment_theses (
  id bigint generated by default as identity primary key,

  user_id uuid not null
    references auth.users(id) on delete cascade,

  stock_id bigint not null
    references public.stocks(id) on delete cascade,

  thesis text not null,

  strengthening_conditions text,
  weakening_conditions text,
  invalidation_conditions text,

  is_active boolean not null default true,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);


-- ============================================================
-- 6. Journal Stocks
-- 특정 날짜의 일지에서 다룬 종목
-- ============================================================

create table if not exists public.journal_stocks (
  id bigint generated by default as identity primary key,

  journal_id bigint not null
    references public.journal_entries(id) on delete cascade,

  stock_id bigint not null
    references public.stocks(id) on delete cascade,

  price numeric,
  change_percent numeric,

  news_impact text,
  short_term_impact text,
  long_term_impact text,

  thesis_change text,

  created_at timestamptz not null default now(),

  constraint journal_stocks_news_impact_check
    check (
      news_impact is null
      or news_impact in ('긍정', '중립', '부정')
    ),

  constraint journal_stocks_thesis_change_check
    check (
      thesis_change is null
      or thesis_change in ('강화', '유지', '약화')
    ),

  constraint journal_stocks_unique
    unique (journal_id, stock_id)
);


-- ============================================================
-- 7. News
-- 오늘의 핵심 뉴스
-- ============================================================

create table if not exists public.news (
  id bigint generated by default as identity primary key,

  user_id uuid not null
    references auth.users(id) on delete cascade,

  journal_id bigint not null
    references public.journal_entries(id) on delete cascade,

  stock_id bigint
    references public.stocks(id) on delete set null,

  title text not null,
  summary text,

  source_name text,
  source_url text,

  fact text,
  interpretation text,
  judgment text,

  importance smallint,

  created_at timestamptz not null default now(),

  constraint news_importance_check
    check (
      importance is null
      or importance between 1 and 5
    )
);


-- ============================================================
-- 8. Investment Decisions
-- 일지에서 내린 투자 판단의 상세 기록
-- ============================================================

create table if not exists public.investment_decisions (
  id bigint generated by default as identity primary key,

  user_id uuid not null
    references auth.users(id) on delete cascade,

  journal_id bigint not null
    references public.journal_entries(id) on delete cascade,

  stock_id bigint
    references public.stocks(id) on delete set null,

  action text not null,

  reason text,
  confidence smallint,

  created_at timestamptz not null default now(),

  constraint investment_decisions_action_check
    check (
      action in (
        '매수',
        '추가매수',
        '보유',
        '관망',
        '일부매도',
        '매도'
      )
    ),

  constraint investment_decisions_confidence_check
    check (
      confidence is null
      or confidence between 1 and 5
    )
);


-- ============================================================
-- 9. Tomorrow Checks
-- 내일 확인할 사항
-- ============================================================

create table if not exists public.tomorrow_checks (
  id bigint generated by default as identity primary key,

  user_id uuid not null
    references auth.users(id) on delete cascade,

  journal_id bigint not null
    references public.journal_entries(id) on delete cascade,

  content text not null,
  is_completed boolean not null default false,

  created_at timestamptz not null default now()
);


-- ============================================================
-- 10. Thesis Reviews
-- 시간이 지나면서 투자 가설이 어떻게 변했는지 기록
-- ============================================================

create table if not exists public.thesis_reviews (
  id bigint generated by default as identity primary key,

  user_id uuid not null
    references auth.users(id) on delete cascade,

  thesis_id bigint not null
    references public.investment_theses(id) on delete cascade,

  journal_id bigint
    references public.journal_entries(id) on delete set null,

  change_type text not null,
  reason text,

  created_at timestamptz not null default now(),

  constraint thesis_reviews_change_type_check
    check (
      change_type in ('강화', '유지', '약화', '재검토')
    )
);


-- ============================================================
-- 11. Indexes
-- 조회 성능을 위한 인덱스
-- ============================================================

create index if not exists idx_journal_entries_user_date
  on public.journal_entries(user_id, journal_date desc);

create index if not exists idx_market_indicators_journal
  on public.market_indicators(journal_id);

create index if not exists idx_stocks_user
  on public.stocks(user_id);

create index if not exists idx_investment_theses_user
  on public.investment_theses(user_id);

create index if not exists idx_investment_theses_stock
  on public.investment_theses(stock_id);

create index if not exists idx_journal_stocks_journal
  on public.journal_stocks(journal_id);

create index if not exists idx_journal_stocks_stock
  on public.journal_stocks(stock_id);

create index if not exists idx_news_user
  on public.news(user_id);

create index if not exists idx_news_journal
  on public.news(journal_id);

create index if not exists idx_investment_decisions_user
  on public.investment_decisions(user_id);

create index if not exists idx_investment_decisions_journal
  on public.investment_decisions(journal_id);

create index if not exists idx_tomorrow_checks_journal
  on public.tomorrow_checks(journal_id);

create index if not exists idx_thesis_reviews_thesis
  on public.thesis_reviews(thesis_id);
